% Pour le code (permet de rajouter un [titre] et {le nom du language de programmation})
\newtcblisting{tcb-code}[2][]{
    listing only,
    breakable,
    colback=tcb-code-back,
    colframe=tcb-code-frame,
    listing options={
        language={#2},
        breaklines=true,
        breakatwhitespace=false,
        keepspaces=true,
        columns=flexible,
        tabsize=4,
        basicstyle=\ttfamily\small,
        keywordstyle=\color{blue},
        commentstyle=\color{gray},
        stringstyle=\color{orange},
        autogobble,
        numbers=left,
        numberstyle=\small\color{gray}\ttfamily,
        numbersep=10pt,
        xleftmargin=2em,
    },
    #1
}
